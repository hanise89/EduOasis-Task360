   351→   351→  const isOwn = initiative.userId === currentUserId;
   352→   352→  const canVerify = isGM && initiative.status === "Dihantar";
   353→   353→
   354→   354→  const hasDetails = Boolean(initiative.problem || initiative.action || initiative.outcome);
