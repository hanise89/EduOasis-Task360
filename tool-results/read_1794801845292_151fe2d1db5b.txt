   361→function daysUntil(dateStr: string) {
   362→  const due = new Date(dateStr);
   363→  const now = new Date();
   364→  now.setHours(0, 0, 0, 0);
   365→  due.setHours(0, 0, 0, 0);
   366→  return Math.round((due.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   367→}
