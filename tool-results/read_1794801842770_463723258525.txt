   145→   145→function exportToCSV(filename: string, rows: string[][]) {
   146→   146→  const csv = rows
   147→   147→    .map((row) =>
   148→   148→      row
   149→   149→        .map((cell) => {
   150→   150→          const s = String(cell ?? "");
   151→   151→          if (s.includes(",") || s.includes('"') || s.includes("\n")) {
   152→   152→            return `"${s.replace(/"/g, '""')}"`;
   153→   153→          }
   154→   154→          return s;
   155→   155→        })
   156→   156→        .join(",")
   157→   157→    )
   158→   158→    .join("\n");
   159→   159→  const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8;" });
   160→   160→  const url = URL.createObjectURL(blob);
   161→   161→  const a = document.createElement("a");
   162→   162→  a.href = url;
   163→   163→  a.download = filename;
   164→   164→  document.body.appendChild(a);
   165→   165→  a.click();
   166→   166→  document.body.removeChild(a);
   167→   167→  URL.revokeObjectURL(url);
   168→   168→}

   653→   653→function KpiTile({
   654→   654→  label,
   655→   655→  value,
   656→   656→  hint,
   657→   657→  icon: Icon,
   658→   658→  accent,
   659→   659→}: {
   660→   660→  label: string;
   661→   661→  value: string | number;
   662→   662→  hint?: string;
   663→   663→  icon: React.ComponentType<{ className?: string }>;
   664→   664→  accent: "primary" | "accent" | "success" | "warning" | "danger" | "neutral";
   665→   665→}) {
   666→   666→  const accentMap: Record<string, { bg: string; text: string; bar: string }> = {
   667→   667→    primary: { bg: "bg-primary/10", text: "text-primary", bar: "bg-primary" },
   668→   668→    accent: { bg: "bg-accent/10", text: "text-accent", bar: "bg-accent" },
   669→   669→    success: { bg: "bg-status-success/10", text: "text-status-success", bar: "bg-status-success" },
   670→   670→    warning: { bg: "bg-status-warning/15", text: "text-status-warning", bar: "bg-status-warning" },
   671→   671→    danger: { bg: "bg-status-danger/10", text: "text-status-danger", bar: "bg-status-danger" },
   672→   672→    neutral: { bg: "bg-muted", text: "text-muted-foreground", bar: "bg-muted-foreground" },
   673→   673→  };
   674→   674→  const a = accentMap[accent];
   675→   675→  return (
   676→   676→    <Card className="p-4 card-flat relative overflow-hidden">
   677→   677→      <div className="flex items-start justify-between gap-2">
   678→   678→        <div className="min-w-0">
   679→   679→          <div className="text-[10px] lg:text-[11px] font-medium uppercase tracking-wide text-muted-foreground truncate">{label}</div>
   680→   680→          <motion.div
   681→   681→            initial={{ opacity: 0, y: 6 }}
   682→   682→            animate={{ opacity: 1, y: 0 }}
   683→   683→            transition={{ duration: 0.35 }}
   684→   684→            className="text-2xl lg:text-3xl font-semibold tracking-tight mt-1.5 tabular-nums"
   685→   685→          >
   686→   686→            {value}
   687→   687→          </motion.div>
   688→   688→          {hint && <div className="text-[11px] text-muted-foreground mt-1">{hint}</div>}
   689→   689→        </div>
   690→   690→        <div className={cn("w-9 h-9 rounded-lg flex items-center justify-center shrink-0", a.bg, a.text)}>
