   361→   361→function daysUntil(dateStr: string) {
   362→   362→  const due = new Date(dateStr);
   363→   363→  const now = new Date();
   364→   364→  now.setHours(0, 0, 0, 0);
   365→   365→  due.setHours(0, 0, 0, 0);
   366→   366→  return Math.round((due.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   367→   367→}
