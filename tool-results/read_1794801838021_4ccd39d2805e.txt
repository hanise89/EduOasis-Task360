   145→function exportToCSV(filename: string, rows: string[][]) {
   146→  const csv = rows
   147→    .map((row) =>
   148→      row
   149→        .map((cell) => {
   150→          const s = String(cell ?? "");
   151→          if (s.includes(",") || s.includes('"') || s.includes("\n")) {
   152→            return `"${s.replace(/"/g, '""')}"`;
   153→          }
   154→          return s;
   155→        })
   156→        .join(",")
   157→    )
   158→    .join("\n");
   159→  const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8;" });
   160→  const url = URL.createObjectURL(blob);
   161→  const a = document.createElement("a");
   162→  a.href = url;
   163→  a.download = filename;
   164→  document.body.appendChild(a);
   165→  a.click();
   166→  document.body.removeChild(a);
   167→  URL.revokeObjectURL(url);
   168→}

   255→function escapeHtml(s: string): string {
   256→  return String(s ?? "")
   257→    .replace(/&/g, "&amp;")
   258→    .replace(/</g, "&lt;")
   259→    .replace(/>/g, "&gt;")
   260→    .replace(/"/g, "&quot;")
   261→    .replace(/'/g, "&#039;");
   262→}

   388→function TeamReportSection({ period }: { period: PeriodKey }) {
   389→  const { data, isLoading, isError } = useQuery({
   390→    queryKey: ["reports", "team", period],
   391→    queryFn: () => apiGet<TeamReport>(`/api/reports/team?period=${period}`),
   392→  });
   393→
   394→  if (isLoading) {
   395→    return (
   396→      <div className="space-y-4">
   397→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   398→          {[...Array(4)].map((_, i) => <LoadingCard key={i} />)}
   399→        </div>
   400→        <Card className="p-4 lg:p-5">
   401→          <Skeleton className="h-6 w-48 mb-4" />
   402→          {[...Array(5)].map((_, i) => <Skeleton key={i} className="h-12 w-full mb-2" />)}
   403→        </Card>
   404→      </div>
   405→    );
   406→  }
   407→
   408→  if (isError) {
   409→    return (
   410→      <Card className="p-6">
   411→        <EmptyState
   412→          icon={AlertTriangle}
   413→          title="Gagal memuatkan laporan pasukan"
   414→          description="Sila cuba lagi. Pastikan anda log masuk sebagai GM."
   415→        />
   416→      </Card>
   417→    );
   418→  }
   419→
   420→  const scores = data?.scores || [];
   421→  const staff = data?.staff || [];
   422→
   423→  if (scores.length === 0) {
   424→    return (
   425→      <Card className="p-6">
   426→        <EmptyState
   427→          icon={Users}
   428→          title="Tiada staf ditemui"
   429→          description="Belum ada staf yang didaftarkan untuk laporan pasukan."
   430→        />
   431→      </Card>
   432→    );
   433→  }
   434→
   435→  // KPIs
   436→  const avgScore = Math.round(scores.reduce((s, x) => s + x.overall, 0) / scores.length);
   437→  const totalStaff = staff.length;
   438→  const totalCompleted = staff.reduce((s, x) => s + (x.tasks?.completed || 0), 0);
   439→  const totalOverdue = staff.reduce((s, x) => s + (x.tasks?.overdue || 0), 0);
   440→
   441→  // Merge staff with score (preserve staff order; rank by score desc)
   442→  const ranked = [...scores].sort((a, b) => b.overall - a.overall);
   443→  const rankMap = new Map<string, number>();
   444→  ranked.forEach((s, i) => rankMap.set(s.userId, i + 1));
   445→
   446→  const rows = staff.map((st) => {
   447→    const sc = scores.find((s) => s.userId === st.id);
   448→    return { staff: st, score: sc, rank: sc ? rankMap.get(st.id) || 0 : 0 };
   449→  });
   450→
   451→  const handleExportCSV = () => {
   452→    const rows: string[][] = [
   453→      ["Kedudukan", "Nama", "Jawatan", "Skor Keseluruhan", "Gred", "Label Gred", "Tugasan Siap", "Tugasan Tertunggak", "Tugasan Aktif", "Jumlah Tugasan", "Trend"],
   454→      ...staff.map((st) => {
   455→        const sc = scores.find((s) => s.userId === st.id);
   456→        const t = st.tasks || { completed: 0, overdue: 0, active: 0, total: 0 };
   457→        const rank = sc ? rankMap.get(st.id) || "" : "";
   458→        const trend = sc ? (sc.trend === "up" ? "Meningkat" : sc.trend === "down" ? "Menurun" : "Stabil") : "";
   459→        return [
   460→          String(rank),
   461→          st.name,
   462→          st.position,
   463→          sc ? String(sc.overall) : "—",
   464→          sc?.grade || "—",
   465→          sc?.gradeLabel || "—",
   466→          String(t.completed),
   467→          String(t.overdue),
   468→          String(t.active),
   469→          String(t.total),
   470→          trend,
   471→        ];
   472→      }),
   473→    ];
   474→    exportToCSV(`laporan-pasukan-${period}-${new Date().toISOString().slice(0, 10)}.csv`, rows);
   475→    toast.success("CSV pasukan telah dimuat turun.");
   476→  };
   477→

   652→
   653→function KpiTile({
   654→  label,
   655→  value,
   656→  hint,
   657→  icon: Icon,
   658→  accent,
   659→}: {
   660→  label: string;
   661→  value: string | number;
   662→  hint?: string;
   663→  icon: React.ComponentType<{ className?: string }>;
   664→  accent: "primary" | "accent" | "success" | "warning" | "danger" | "neutral";
   665→}) {
   666→  const accentMap: Record<string, { bg: string; text: string; bar: string }> = {
   667→    primary: { bg: "bg-primary/10", text: "text-primary", bar: "bg-primary" },
   668→    accent: { bg: "bg-accent/10", text: "text-accent", bar: "bg-accent" },
   669→    success: { bg: "bg-status-success/10", text: "text-status-success", bar: "bg-status-success" },
   670→    warning: { bg: "bg-status-warning/15", text: "text-status-warning", bar: "bg-status-warning" },
   671→    danger: { bg: "bg-status-danger/10", text: "text-status-danger", bar: "bg-status-danger" },
   672→    neutral: { bg: "bg-muted", text: "text-muted-foreground", bar: "bg-muted-foreground" },
   673→  };
   674→  const a = accentMap[accent];
   675→  return (
   676→    <Card className="p-4 card-flat relative overflow-hidden">
   677→      <div className="flex items-start justify-between gap-2">
   678→        <div className="min-w-0">
   679→          <div className="text-[10px] lg:text-[11px] font-medium uppercase tracking-wide text-muted-foreground truncate">{label}</div>
   680→          <motion.div
   681→            initial={{ opacity: 0, y: 6 }}
   682→            animate={{ opacity: 1, y: 0 }}
   683→            transition={{ duration: 0.35 }}
   684→            className="text-2xl lg:text-3xl font-semibold tracking-tight mt-1.5 tabular-nums"
   685→          >
   686→            {value}
   687→          </motion.div>
   688→          {hint && <div className="text-[11px] text-muted-foreground mt-1">{hint}</div>}
   689→        </div>
   690→        <div className={cn("w-9 h-9 rounded-lg flex items-center justify-center shrink-0", a.bg, a.text)}>
   691→          <Icon className="w-4 h-4" />
   692→        </div>
   693→      </div>
   694→      <div className={cn("absolute inset-x-0 top-0 h-0.5", a.bar)} />
   695→    </Card>
   696→  );
   697→}
