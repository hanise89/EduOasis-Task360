   190→   190→function WeightsTab() {
   191→   191→  const qc = useQueryClient();
   192→   192→  const { data, isLoading } = useQuery({
   193→   193→    queryKey: ["settings"],
   194→   194→    queryFn: () => apiGet<{ settings: Settings }>("/api/settings"),
   195→   195→  });
   196→   196→
   197→   197→  const [weights, setWeights] = useState<Settings["weights"]>(
   198→   198→    DEFAULT_SETTINGS.weights
   199→   199→  );
   200→   200→  const [saving, setSaving] = useState(false);
   201→   201→  const [hydrated, setHydrated] = useState(false);
   202→   202→
   203→   203→  useEffect(() => {
   204→   204→    if (data?.settings?.weights) {
   205→   205→      setWeights(data.settings.weights);
   206→   206→      setHydrated(true);
   207→   207→    }
   208→   208→  }, [data]);
   209→   209→
   210→   210→  const total = useMemo(
   211→   211→    () =>
   212→   212→      Object.values(weights).reduce(
   213→   213→        (sum, v) => sum + (Number(v) || 0),
   214→   214→        0
   215→   215→      ),
   216→   216→    [weights]
   217→   217→  );
   218→   218→
   219→   219→  const valid = Math.abs(total - 100) <= 0.5;
   220→   220→  const dirty =
   221→   221→    !hydrated ||
   222→   222→    JSON.stringify(data?.settings?.weights ?? DEFAULT_SETTINGS.weights) !==
   223→   223→      JSON.stringify(weights);
   224→   224→
   225→   225→  async function handleSave() {
   226→   226→    if (!valid) {
   227→   227→      toast.error("Jumlah pemberat mesti 100%.", {
   228→   228→        description: `Kini ${total}%.`,
   229→   229→      });
   230→   230→      return;
   231→   231→    }
   232→   232→    setSaving(true);
   233→   233→    try {
   234→   234→      const payload: Settings = {
   235→   235→        ...(data?.settings ?? DEFAULT_SETTINGS),
   236→   236→        weights,
   237→   237→      };
   238→   238→      const res = await apiPut<{ settings: Settings }>(
   239→   239→        "/api/settings",
   240→   240→        payload
   241→   241→      );
   242→   242→      setWeights(res.settings.weights);
   243→   243→      qc.invalidateQueries({ queryKey: ["settings"] });
   244→   244→      qc.invalidateQueries({ queryKey: ["performance"] });
   245→   245→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   246→   246→      toast.success("Pemberat dimensi disimpan.");
   247→   247→    } catch (e: unknown) {
   248→   248→      const msg = e instanceof Error ? e.message : "Ralat tidak diketahui";
   249→   249→      toast.error("Gagal menyimpan pemberat.", { description: msg });
   250→   250→    } finally {
   251→   251→      setSaving(false);
   252→   252→    }
   253→   253→  }
   254→   254→
   255→   255→  if (isLoading) return <WeightsSkeleton />;
   256→   256→
   257→   257→  return (
   258→   258→    <Card className="p-4 lg:p-6">
   259→   259→      <SectionHeader

   263→   263→        action={
   264→   264→          <Button
   265→   265→            size="sm"
   266→   266→            onClick={handleSave}
   267→   267→            disabled={!valid || saving || !dirty}
   268→   268→          >
   269→   269→            {saving ? (
   270→   270→              <Loader2 className="w-4 h-4 animate-spin" />
   271→   271→            ) : (
   272→   272→              <Save className="w-4 h-4" />
   273→   273→            )}
   274→   274→            Simpan Pemberat
   275→   275→          </Button>
   276→   276→        }
   277→   277→      />
   278→   278→
   279→   279→      {/* Total bar */}
   280→   280→      <div
   281→   281→        className={cn(
   282→   282→          "flex items-center justify-between gap-3 rounded-lg border px-4 py-3 mb-4",
   283→   283→          valid ? "border-border bg-muted/40" : "border-status-danger/30 bg-status-danger/5"
   284→   284→        )}
   285→   285→      >
   286→   286→        <div className="flex items-center gap-2">
   287→   287→          {valid ? (
   288→   288→            <CheckCircle2 className="w-4 h-4 text-status-success" />
   289→   289→          ) : (
   290→   290→            <AlertTriangle className="w-4 h-4 text-status-danger" />

   295→   295→          <span
   296→   296→            className={cn(
   297→   297→              "text-2xl font-semibold tabular-nums",
   298→   298→              valid ? "text-foreground" : "text-status-danger"
   299→   299→            )}

   301→   301→            {total}%

   320→   320→        {DIMENSIONS.map((d) => {
   321→   321→          const val = Number(weights[d.key]) || 0;
   322→   322→          return (
   323→   323→            <div

   380→   380→function TargetsTab() {

   382→   382→  const { data, isLoading } = useQuery({
   383→   383→    queryKey: ["settings"],
   384→   384→    queryFn: () => apiGet<{ settings: Settings }>("/api/settings"),
   385→   385→  });
   386→   386→
   387→   387→  const [form, setForm] = useState<Settings>(DEFAULT_SETTINGS);
   388→   388→  const [saving, setSaving] = useState(false);
   389→   389→  const [hydrated, setHydrated] = useState(false);
   390→   390→  const [newHoliday, setNewHoliday] = useState("");
   391→   391→
   392→   392→  useEffect(() => {
   393→   393→    if (data?.settings) {
   394→   394→      setForm(data.settings);
   395→   395→      setHydrated(true);
   396→   396→    }
   397→   397→  }, [data]);
   398→   398→
   399→   399→  const dirty =
   400→   400→    !hydrated ||
   401→   401→    JSON.stringify(data?.settings ?? DEFAULT_SETTINGS) !== JSON.stringify(form);
   402→   402→
   403→   403→  function setField<K extends keyof Settings>(key: K, value: Settings[K]) {
   404→   404→    setForm((f) => ({ ...f, [key]: value }));
   405→   405→  }
   406→   406→
   407→   407→  function setGrade(
   408→   408→    grade: keyof Settings["gradeBands"],
   409→   409→    field: "min" | "max",
   410→   410→    value: number
   411→   411→  ) {
   412→   412→    setForm((f) => ({
   413→   413→      ...f,
   414→   414→      gradeBands: {
   415→   415→        ...f.gradeBands,
   416→   416→        [grade]: { ...f.gradeBands[grade], [field]: value },
   417→   417→      },
   418→   418→    }));
   419→   419→  }
   420→   420→
   421→   421→  function toggleWorkingDay(day: number) {
   422→   422→    setForm((f) => ({
   423→   423→      ...f,
   424→   424→      workingDays: f.workingDays.includes(day)
   425→   425→        ? f.workingDays.filter((d) => d !== day)
   426→   426→        : [...f.workingDays, day].sort((a, b) => a - b),
   427→   427→    }));
   428→   428→  }
   429→   429→
   430→   430→  function addHoliday() {
   431→   431→    if (!newHoliday) return;
   432→   432→    if (form.holidays.includes(newHoliday)) {
   433→   433→      toast.error("Tarikh cuti sudah wujud dalam senarai.");
   434→   434→      return;
   435→   435→    }
   436→   436→    setForm((f) => ({
   437→   437→      ...f,
   438→   438→      holidays: [...f.holidays, newHoliday].sort(),
   439→   439→    }));
   440→   440→    setNewHoliday("");
   441→   441→  }
   442→   442→
   443→   443→  function removeHoliday(date: string) {
   444→   444→    setForm((f) => ({
   445→   445→      ...f,
   446→   446→      holidays: f.holidays.filter((d) => d !== date),
   447→   447→    }));
   448→   448→  }
   449→   449→
   450→   450→  async function handleSave() {
   451→   451→    setSaving(true);
   452→   452→    try {
   453→   453→      const res = await apiPut<{ settings: Settings }>("/api/settings", form);
   454→   454→      setForm(res.settings);
   455→   455→      qc.invalidateQueries({ queryKey: ["settings"] });
   456→   456→      qc.invalidateQueries({ queryKey: ["performance"] });
   457→   457→      qc.invalidateQueries({ queryKey: ["workload"] });
   458→   458→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   459→   459→      toast.success("Tetapan sasaran & gred disimpan.");
   460→   460→    } catch (e: unknown) {
   461→   461→      const msg = e instanceof Error ? e.message : "Ralat tidak diketahui";
   462→   462→      toast.error("Gagal menyimpan tetapan.", { description: msg });
   463→   463→    } finally {
   464→   464→      setSaving(false);
   465→   465→    }
   466→   466→  }
   467→   467→
   468→   468→  if (isLoading) return <TargetsSkeleton />;
   469→   469→
   470→   470→  return (
   471→   471→    <div className="space-y-4">
   472→   472→      {/* Initiative targets */}
   473→   473→      <Card className="p-4 lg:p-6">
   474→   474→        <SectionHeader
   475→   475→          title="Sasaran Inisiatif"
   476→   476→          subtitle="Sasaran mata inisiatif bulanan & suku tahunan"
   477→   477→          icon={Award}
   478→   478→        />
