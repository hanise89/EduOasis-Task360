   190→function WeightsTab() {
   191→  const qc = useQueryClient();
   192→  const { data, isLoading } = useQuery({
   193→    queryKey: ["settings"],
   194→    queryFn: () => apiGet<{ settings: Settings }>("/api/settings"),
   195→  });
   196→
   197→  const [weights, setWeights] = useState<Settings["weights"]>(
   198→    DEFAULT_SETTINGS.weights
   199→  );
   200→  const [saving, setSaving] = useState(false);
   201→  const [hydrated, setHydrated] = useState(false);
   202→
   203→  useEffect(() => {
   204→    if (data?.settings?.weights) {
   205→      setWeights(data.settings.weights);
   206→      setHydrated(true);
   207→    }
   208→  }, [data]);
   209→
   210→  const total = useMemo(
   211→    () =>
   212→      Object.values(weights).reduce(
   213→        (sum, v) => sum + (Number(v) || 0),
   214→        0
   215→      ),
   216→    [weights]
   217→  );
   218→
   219→  const valid = Math.abs(total - 100) <= 0.5;
   220→  const dirty =
   221→    !hydrated ||
   222→    JSON.stringify(data?.settings?.weights ?? DEFAULT_SETTINGS.weights) !==
   223→      JSON.stringify(weights);
   224→
   225→  async function handleSave() {
   226→    if (!valid) {
   227→      toast.error("Jumlah pemberat mesti 100%.", {
   228→        description: `Kini ${total}%.`,
   229→      });
   230→      return;
   231→    }
   232→    setSaving(true);
   233→    try {
   234→      const payload: Settings = {
   235→        ...(data?.settings ?? DEFAULT_SETTINGS),
   236→        weights,
   237→      };
   238→      const res = await apiPut<{ settings: Settings }>(
   239→        "/api/settings",
   240→        payload
   241→      );
   242→      setWeights(res.settings.weights);
   243→      qc.invalidateQueries({ queryKey: ["settings"] });
   244→      qc.invalidateQueries({ queryKey: ["performance"] });
   245→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   246→      toast.success("Pemberat dimensi disimpan.");
   247→    } catch (e: unknown) {
   248→      const msg = e instanceof Error ? e.message : "Ralat tidak diketahui";
   249→      toast.error("Gagal menyimpan pemberat.", { description: msg });
   250→    } finally {
   251→      setSaving(false);
   252→    }
   253→  }
   254→
   255→  if (isLoading) return <WeightsSkeleton />;
   256→
   257→  return (
   258→    <Card className="p-4 lg:p-6">
   259→      <SectionHeader
   260→        title="Pemberat Dimensi Prestasi"
   261→        subtitle="Jumlah pemberat keenam-enam dimensi mesti 100%"
   262→        icon={Scale}
   263→        action={
   264→          <Button
   265→            size="sm"
   266→            onClick={handleSave}
   267→            disabled={!valid || saving || !dirty}
   268→          >
   269→            {saving ? (
   270→              <Loader2 className="w-4 h-4 animate-spin" />
   271→            ) : (
   272→              <Save className="w-4 h-4" />
   273→            )}
   274→            Simpan Pemberat
   275→          </Button>
   276→        }
   277→      />
   278→
   279→      {/* Total bar */}
   280→      <div
   281→        className={cn(
   282→          "flex items-center justify-between gap-3 rounded-lg border px-4 py-3 mb-4",
   283→          valid ? "border-border bg-muted/40" : "border-status-danger/30 bg-status-danger/5"
   284→        )}
   285→      >
   286→        <div className="flex items-center gap-2">
   287→          {valid ? (
   288→            <CheckCircle2 className="w-4 h-4 text-status-success" />
   289→          ) : (
   290→            <AlertTriangle className="w-4 h-4 text-status-danger" />
   291→          )}
   292→          <span className="text-sm font-medium">Jumlah Pemberat</span>
   293→        </div>
   294→        <div className="flex items-center gap-3">
   295→          <span
   296→            className={cn(
   297→              "text-2xl font-semibold tabular-nums",
   298→              valid ? "text-foreground" : "text-status-danger"
   299→            )}
   300→          >
   301→            {total}%
   302→          </span>
   303→          <Badge variant={valid ? "secondary" : "destructive"}>
   304→            {valid ? "Sah" : `Baki ${100 - total}%`}
   305→          </Badge>
   306→        </div>
   307→      </div>
   308→
   309→      {!valid && (
   310→        <div className="flex items-start gap-2 rounded-md bg-status-danger/10 border border-status-danger/20 px-3 py-2 mb-4 text-xs text-status-danger">
   311→          <AlertTriangle className="w-4 h-4 shrink-0 mt-0.5" />
   312→          <span>
   313→            Jumlah pemberat kini <strong>{total}%</strong>. Laraskan supaya
   314→            berjumlah 100% untuk membolehkan butang Simpan.
   315→          </span>
   316→        </div>
   317→      )}
   318→
   319→      <div className="space-y-5">
   320→        {DIMENSIONS.map((d) => {
   321→          const val = Number(weights[d.key]) || 0;
   322→          return (
   323→            <div
   324→              key={d.key}
   325→              className="grid grid-cols-1 sm:grid-cols-[1fr_auto] gap-3 sm:gap-4 items-start sm:items-center"
   326→            >
   327→              <div className="min-w-0">
   328→                <div className="flex items-baseline justify-between gap-3 mb-1.5">
   329→                  <div className="min-w-0">
   330→                    <Label className="text-sm font-medium">{d.label}</Label>
   331→                    <p className="text-[11px] text-muted-foreground truncate">
   332→                      {d.desc}
   333→                    </p>
   334→                  </div>
   335→                  <span className="text-sm font-semibold tabular-nums text-accent shrink-0">
   336→                    {val}%
   337→                  </span>
   338→                </div>
   339→                <Slider
   340→                  value={[val]}
   341→                  min={0}
   342→                  max={100}
   343→                  step={1}
   344→                  onValueChange={(v) =>
   345→                    setWeights((w) => ({ ...w, [d.key]: v[0] ?? 0 }))
   346→                  }
   347→                  className="w-full"
   348→                  aria-label={d.label}
   349→                />
   350→              </div>
   351→              <div className="flex items-center gap-2 sm:w-28">
   352→                <Input
   353→                  type="number"
   354→                  min={0}
   355→                  max={100}
   356→                  value={val}
   357→                  onChange={(e) =>
   358→                    setWeights((w) => ({
   359→                      ...w,
   360→                      [d.key]: Math.max(
   361→                        0,
   362→                        Math.min(100, Number(e.target.value) || 0)
   363→                      ),
   364→                    }))
   365→                  }
   366→                  className="text-right tabular-nums"
   367→                />
   368→                <span className="text-xs text-muted-foreground">%</span>
   369→              </div>
   370→            </div>
   371→          );
   372→        })}
   373→      </div>
   374→    </Card>
   375→  );
   376→}

   378→// --- Tab 2: Targets & Grades ----------------------------------------------
   379→
   380→function TargetsTab() {
   381→  const qc = useQueryClient();
   382→  const { data, isLoading } = useQuery({
   383→    queryKey: ["settings"],
   384→    queryFn: () => apiGet<{ settings: Settings }>("/api/settings"),
   385→  });
   386→
   387→  const [form, setForm] = useState<Settings>(DEFAULT_SETTINGS);
   388→  const [saving, setSaving] = useState(false);
   389→  const [hydrated, setHydrated] = useState(false);
   390→  const [newHoliday, setNewHoliday] = useState("");
   391→
   392→  useEffect(() => {
   393→    if (data?.settings) {
   394→      setForm(data.settings);
   395→      setHydrated(true);
   396→    }
   397→  }, [data]);
   398→
   399→  const dirty =
   400→    !hydrated ||
   401→    JSON.stringify(data?.settings ?? DEFAULT_SETTINGS) !== JSON.stringify(form);
   402→
   403→  function setField<K extends keyof Settings>(key: K, value: Settings[K]) {
   404→    setForm((f) => ({ ...f, [key]: value }));
   405→  }
   406→
   407→  function setGrade(
   408→    grade: keyof Settings["gradeBands"],
   409→    field: "min" | "max",
   410→    value: number
   411→  ) {
   412→    setForm((f) => ({
   413→      ...f,
   414→      gradeBands: {
   415→        ...f.gradeBands,
   416→        [grade]: { ...f.gradeBands[grade], [field]: value },
   417→      },
   418→    }));
   419→  }
   420→
   421→  function toggleWorkingDay(day: number) {
   422→    setForm((f) => ({
   423→      ...f,
   424→      workingDays: f.workingDays.includes(day)
   425→        ? f.workingDays.filter((d) => d !== day)
   426→        : [...f.workingDays, day].sort((a, b) => a - b),
   427→    }));
   428→  }
   429→
   430→  function addHoliday() {
   431→    if (!newHoliday) return;
   432→    if (form.holidays.includes(newHoliday)) {
   433→      toast.error("Tarikh cuti sudah wujud dalam senarai.");
   434→      return;
   435→    }
   436→    setForm((f) => ({
   437→      ...f,
   438→      holidays: [...f.holidays, newHoliday].sort(),
   439→    }));
   440→    setNewHoliday("");
   441→  }
   442→
   443→  function removeHoliday(date: string) {
   444→    setForm((f) => ({
   445→      ...f,
   446→      holidays: f.holidays.filter((d) => d !== date),
   447→    }));
   448→  }
   449→
   450→  async function handleSave() {
   451→    setSaving(true);
   452→    try {
   453→      const res = await apiPut<{ settings: Settings }>("/api/settings", form);
   454→      setForm(res.settings);
   455→      qc.invalidateQueries({ queryKey: ["settings"] });
   456→      qc.invalidateQueries({ queryKey: ["performance"] });
   457→      qc.invalidateQueries({ queryKey: ["workload"] });
   458→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   459→      toast.success("Tetapan sasaran & gred disimpan.");
   460→    } catch (e: unknown) {
   461→      const msg = e instanceof Error ? e.message : "Ralat tidak diketahui";
   462→      toast.error("Gagal menyimpan tetapan.", { description: msg });
   463→    } finally {
   464→      setSaving(false);
   465→    }
   466→  }
   467→
   468→  if (isLoading) return <TargetsSkeleton />;
   469→
   470→  return (
   471→    <div className="space-y-4">
   472→      {/* Initiative targets */}
   473→      <Card className="p-4 lg:p-6">
   474→        <SectionHeader
   475→          title="Sasaran Inisiatif"
   476→          subtitle="Sasaran mata inisiatif bulanan & suku tahunan"
   477→          icon={Award}
   478→        />
   479→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   480→          <div className="space-y-1.5">
   481→            <Label htmlFor="monthly-target">Sasaran Bulanan (mata)</Label>
   482→            <Input
   483→              id="monthly-target"
   484→              type="number"
   485→              min={0}
   486→              max={200}
   487→              value={form.initiativeTargetMonthly}
   488→              onChange={(e) =>
   489→                setField(
   490→                  "initiativeTargetMonthly",
   491→                  Math.max(0, Number(e.target.value) || 0)
   492→                )
   493→              }
   494→              className="tabular-nums"
   495→            />
   496→            <p className="text-[11px] text-muted-foreground">
   497→              Minimum mata inisiatif setiap staf sebulan
   498→            </p>
   499→          </div>
   500→          <div className="space-y-1.5">
   501→            <Label htmlFor="quarterly-target">Sasaran Suku Tahunan (mata)</Label>
   502→            <Input
   503→              id="quarterly-target"
   504→              type="number"
   505→              min={0}
   506→              max={500}
   507→              value={form.initiativeTargetQuarterly}
   508→              onChange={(e) =>
   509→                setField(
   510→                  "initiativeTargetQuarterly",
   511→                  Math.max(0, Number(e.target.value) || 0)
   512→                )
   513→              }
   514→              className="tabular-nums"
   515→            />
   516→            <p className="text-[11px] text-muted-foreground">
   517→              Minimum mata inisiatif setiap staf setiap suku tahun
   518→            </p>
   519→          </div>
   520→        </div>
   521→      </Card>
   522→
   523→      {/* Grade bands */}
   524→      <Card className="p-4 lg:p-6">
   525→        <SectionHeader
   526→          title="Julat Gred Prestasi"
   527→          subtitle="Takrif julat markah bagi setiap gred (0-100)"
   528→          icon={Award}
   529→        />
   530→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   531→          {(
   532→            Object.keys(form.gradeBands) as (keyof Settings["gradeBands"])[]
   533→          ).map((g) => {
   534→            const band = form.gradeBands[g];
   535→            const tone =
   536→              g === "A" || g === "B"
   537→                ? "bg-status-success/10 text-status-success"
   538→                : g === "C"
   539→                  ? "bg-status-warning/15 text-status-warning"
   540→                  : "bg-status-danger/10 text-status-danger";
   541→            return (
   542→              <div
   543→                key={g}
   544→                className="rounded-lg border border-border p-3 bg-muted/20"
   545→              >
   546→                <div className="flex items-center justify-between mb-2">
   547→                  <div className="flex items-center gap-2">
   548→                    <span
   549→                      className={cn(
   550→                        "w-7 h-7 rounded-md flex items-center justify-center text-sm font-semibold",
   551→                        tone
   552→                      )}
   553→                    >
   554→                      {g}
   555→                    </span>
   556→                    <span className="text-sm font-medium">
   557→                      {GRADE_LABELS[g]}
   558→                    </span>
   559→                  </div>
   560→                </div>
   561→                <div className="grid grid-cols-2 gap-2">
   562→                  <div>
   563→                    <Label className="text-[11px] text-muted-foreground">
   564→                      Minimum
   565→                    </Label>
   566→                    <Input
   567→                      type="number"
   568→                      min={0}
   569→                      max={100}
   570→                      value={band.min}
   571→                      onChange={(e) =>
   572→                        setGrade(
   573→                          g,
   574→                          "min",
   575→                          Math.max(
   576→                            0,
   577→                            Math.min(100, Number(e.target.value) || 0)
   578→                          )
   579→                        )
   580→                      }
   581→                      className="tabular-nums"
   582→                    />
   583→                  </div>
   584→                  <div>
   585→                    <Label className="text-[11px] text-muted-foreground">
   586→                      Maksimum
   587→                    </Label>
   588→                    <Input
   589→                      type="number"
   590→                      min={0}
