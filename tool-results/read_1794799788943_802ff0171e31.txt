   338→  currentUserId: string;
   339→  isGM: boolean;
   340→  onMutated: () => void;

   348→  const ownerPosition = owner?.position || "";
   349→  const avatarColor = owner?.avatarColor || "#1F3A5F";
   350→
   351→  const isOwn = initiative.userId === currentUserId;
   352→  const canVerify = isGM && initiative.status === "Dihantar";
   353→
   354→  const hasDetails = Boolean(initiative.problem || initiative.action || initiative.outcome);
   355→

   372→              <div className="flex items-center gap-1.5 flex-wrap">

   383→                  <>
   384→                    <span aria-hidden>·</span>

   389→                <span>{relativeTime(initiative.submittedAt)}</span>
   390→              </div>
   391→            </div>
   392→
   393→            <div className="flex items-center gap-1.5 flex-wrap shrink-0">
   394→              <span
   395→                className={cn(
   396→                  "inline-flex items-center gap-1.5 px-2 py-0.5 rounded-md text-xs font-medium",
   397→                  statusMeta.bg,
   398→                  statusMeta.text
   399→                )}
   400→              >
   401→                <span className={cn("w-1.5 h-1.5 rounded-full", statusMeta.dot)} />
   402→                {statusMeta.label}
   403→              </span>
   404→              {initiative.status === "Disahkan" && impactMeta && (
   405→                <span
   406→                  className={cn(
   407→                    "inline-flex items-center gap-1.5 px-2 py-0.5 rounded-md text-xs font-medium",
   408→                    impactMeta.bg,
   409→                    impactMeta.text
   410→                  )}
   411→                  title={`Impak ${impactMeta.label} = ${POINTS[impact]} mata`}
   412→                >
   413→                  <Sparkles className="w-3 h-3" />
   414→                  {impactMeta.label} · {initiative.points} mata
   415→                </span>
   416→              )}
   417→            </div>
   418→          </div>
   419→
   420→          {/* Outcome preview */}
   421→          {initiative.outcome && (
   422→            <p className="text-xs text-muted-foreground mt-2 line-clamp-2">
   423→              {initiative.outcome}
   424→            </p>
   425→          )}
   426→
   427→          {/* Action row */}
   428→          <div className="flex items-center gap-2 mt-2 flex-wrap">
   429→            {hasDetails && (
   430→              <button
   431→                onClick={() => setExpanded((v) => !v)}
   432→                className="inline-flex items-center gap-1 text-xs font-medium text-accent hover:underline"
   433→              >
   434→                {expanded ? (
   435→                  <ChevronDown className="w-3.5 h-3.5" />
   436→                ) : (
   437→                  <ChevronRight className="w-3.5 h-3.5" />
   438→                )}
   439→                {expanded ? "Tutup butiran" : "Lihat butiran"}
   440→              </button>
   441→            )}
   442→            {initiative.evidenceLinks && initiative.evidenceLinks.length > 0 && (
   443→              <div className="flex items-center gap-1 flex-wrap">
   444→                {initiative.evidenceLinks.slice(0, 2).map((link, i) => (
   445→                  <a
   446→                    key={i}
   447→                    href={link}
   448→                    target="_blank"
   449→                    rel="noopener noreferrer"

   453→                    <Paperclip className="w-3 h-3 shrink-0" />

   469→      {/* Expandable details */}
   470→      <AnimatePresence initial={false}>
   471→        {expanded && hasDetails && (
   472→          <motion.div
   473→            initial={{ height: 0, opacity: 0 }}
   474→            animate={{ height: "auto", opacity: 1 }}

   479→            <div className="px-3 lg:px-4 pb-3 lg:pb-4 pt-1 grid grid-cols-1 gap-2.5">

   482→              <DetailRow icon={Award} label="Hasil" text={initiative.outcome} />

   484→                <div className="flex items-center gap-1.5 text-xs text-muted-foreground pt-1">
   485→                  <Clock className="w-3.5 h-3.5" />

   490→                <div className="mt-1 p-2.5 rounded-md bg-muted/50 border border-border">

   495→                </div>
   496→              )}
   497→            </div>
   498→          </motion.div>
   499→        )}
   500→      </AnimatePresence>
   501→
   502→      {/* Verify form (GM + Dihantar) */}
   503→      {canVerify && (
   504→        <VerifyForm initiative={initiative} onDone={onMutated} />
   505→      )}
   506→    </div>
   507→  );
   508→}

   533→// ---- Verify form ----------------------------------------------------------
   534→
   535→function VerifyForm({
   536→  initiative,
   537→  onDone,
   538→}: {
   539→  initiative: InitiativeWithUser;
   540→  onDone: () => void;
   541→}) {
   542→  const [impact, setImpact] = useState<ImpactLevel>("Sederhana");
   543→  const [comment, setComment] = useState("");
   544→  const [submitting, setSubmitting] = useState<"Disahkan" | "Ditolak" | null>(null);
   545→
   546→  const handleVerify = async (status: "Disahkan" | "Ditolak") => {
   547→    setSubmitting(status);
   548→    try {
   549→      await apiPatch(`/api/initiatives/${initiative.id}/verify`, {
   550→        status,
   551→        impactLevel: status === "Disahkan" ? impact : initiative.impactLevel,
   552→        gmComment: comment.trim(),
   553→      });
   554→      toast.success(
   555→        status === "Disahkan"
   556→          ? `Inisiatif disahkan (${POINTS[impact]} mata).`
   557→          : "Inisiatif ditolak."
   558→      );
   559→      setComment("");
   560→      onDone();
   561→    } catch (e: any) {
   562→      toast.error(e?.message || "Gagal mengesahkan inisiatif.");
   563→    } finally {
   564→      setSubmitting(null);
   565→    }
   566→  };
   567→
   568→  return (
   569→    <div className="border-t border-border bg-muted/20 p-3 lg:p-4">
   570→      <div className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground mb-2 flex items-center gap-1.5">
   571→        <Award className="w-3.5 h-3.5 text-accent" />
   572→        Pengesahan GM
   573→      </div>
   574→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   575→        <div className="space-y-1.5">
   576→          <Label className="text-xs">Tahap Impak</Label>

   760→
   761→function CreateInitiativeDialog({
   762→  open,
   763→  onOpenChange,
   764→  onCreated,
   765→}: {
   766→  open: boolean;
   767→  onOpenChange: (o: boolean) => void;
   768→  onCreated: () => void;
   769→}) {
   770→  const [title, setTitle] = useState("");
   771→  const [problem, setProblem] = useState("");
   772→  const [action, setAction] = useState("");
   773→  const [outcome, setOutcome] = useState("");
   774→  const [links, setLinks] = useState<string[]>([""]);
   775→  const [submitting, setSubmitting] = useState(false);
   776→
   777→  const reset = () => {
   778→    setTitle("");
   779→    setProblem("");
   780→    setAction("");
   781→    setOutcome("");
   782→    setLinks([""]);
   783→  };
   784→
   785→  const handleClose = (o: boolean) => {
   786→    onOpenChange(o);
   787→    if (!o) reset();
   788→  };
   789→
   790→  const addLink = () => setLinks((l) => [...l, ""]);
   791→  const updateLink = (i: number, v: string) =>
   792→    setLinks((l) => l.map((link, idx) => (idx === i ? v : link)));
   793→  const removeLink = (i: number) =>
   794→    setLinks((l) => (l.length > 1 ? l.filter((_, idx) => idx !== i) : l));
   795→
   796→  const handleSubmit = async () => {
   797→    if (!title.trim()) {
   798→      toast.error("Sila isi tajuk inisiatif.");
   799→      return;
   800→    }
   801→    setSubmitting(true);
   802→    try {
   803→      const evidenceLinks = links.map((l) => l.trim()).filter(Boolean);
   804→      await apiPost("/api/initiatives", {
   805→        title: title.trim(),
   806→        problem: problem.trim(),
   807→        action: action.trim(),
   808→        outcome: outcome.trim(),
   809→        evidenceLinks,
   810→      });
   811→      toast.success("Inisiatif dihantar untuk pengesahan GM.");
   812→      handleClose(false);
   813→      onCreated();
   814→    } catch (e: any) {
   815→      toast.error(e?.message || "Gagal menghantar inisiatif.");
   816→    } finally {
   817→      setSubmitting(false);
   818→    }
   819→  };
   820→
   821→  return (
   822→    <Dialog open={open} onOpenChange={handleClose}>
   823→      <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto scroll-slim">
   824→        <DialogHeader>
   825→          <DialogTitle className="flex items-center gap-2">
   826→            <Lightbulb className="w-4 h-4 text-accent" />
   827→            Hantar Inisiatif Baharu
   828→          </DialogTitle>
   829→        </DialogHeader>
   830→
   831→        <div className="space-y-3.5 py-2">
   832→          <div className="space-y-1.5">
   833→            <Label className="text-xs">
   834→              Tajuk Inisiatif <span className="text-status-danger">*</span>
   835→            </Label>
   836→            <Input
   837→              value={title}
   838→              onChange={(e) => setTitle(e.target.value)}
   839→              placeholder="Contoh: Automasi laporan mingguan jualan"
   840→              className="h-9"
   841→            />
   842→          </div>
   843→
   844→          <div className="space-y-1.5">
   845→            <Label className="text-xs">Masalah / Pencabar</Label>
   846→            <Textarea
   847→              value={problem}
   848→              onChange={(e) => setProblem(e.target.value)}
   849→              placeholder="Apakah masalah yang dikenal pasti?"
   850→              rows={2}
   851→            />
   852→          </div>
   853→
   854→          <div className="space-y-1.5">
   855→            <Label className="text-xs">Tindakan</Label>
   856→            <Textarea
   857→              value={action}
   858→              onChange={(e) => setAction(e.target.value)}
   859→              placeholder="Langkah yang diambil..."
   860→              rows={2}
   861→            />
   862→          </div>
   863→
   864→          <div className="space-y-1.5">
   865→            <Label className="text-xs">Hasil / Outcome</Label>
   866→            <Textarea
   867→              value={outcome}
   868→              onChange={(e) => setOutcome(e.target.value)}
   869→              placeholder="Apakah hasil yang dicapai?"
   870→              rows={2}
   871→            />
   872→          </div>
   873→
   874→          <div className="space-y-1.5">
   875→            <div className="flex items-center justify-between">
   876→              <Label className="text-xs">Pautan Bukti (pilihan)</Label>
   877→              <Button
   878→                type="button"
   879→                variant="ghost"
   880→                size="sm"
   881→                className="h-7 px-2 text-xs"
   882→                onClick={addLink}
   883→              >
   884→                <Plus className="w-3.5 h-3.5 mr-1" /> Tambah pautan
   885→              </Button>
   886→            </div>
   887→            <div className="space-y-2">
   888→              {links.map((link, i) => (
   889→                <div key={i} className="flex items-center gap-1.5">
   890→                  <Input
   891→                    value={link}
   892→                    onChange={(e) => updateLink(i, e.target.value)}
   893→                    placeholder="https://..."
   894→                    className="h-9"
   895→                  />
   896→                  {links.length > 1 && (
   897→                    <Button
   898→                      type="button"
   899→                      variant="ghost"
   900→                      size="sm"
   901→                      className="h-9 w-9 p-0 text-muted-foreground hover:text-status-danger"
   902→                      onClick={() => removeLink(i)}
   903→                      aria-label="Buang pautan"
   904→                    >
   905→                      <Trash2 className="w-4 h-4" />
   906→                    </Button>
   907→                  )}
   908→                </div>
   909→              ))}
   910→            </div>
   911→          </div>
   912→        </div>
   913→
   914→        <DialogFooter>
   915→          <Button variant="outline" onClick={() => handleClose(false)} disabled={submitting}>
   916→            Batal
   917→          </Button>
   918→          <Button onClick={handleSubmit} disabled={submitting}>
   919→            {submitting ? (
   920→              <Loader2 className="w-4 h-4 mr-1 animate-spin" />
   921→            ) : (
   922→              <Send className="w-4 h-4 mr-1" />
   923→            )}
   924→            Hantar
   925→          </Button>
   926→        </DialogFooter>
   927→      </DialogContent>
   928→    </Dialog>
   929→  );
   930→}
